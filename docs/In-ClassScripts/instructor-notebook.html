Overview

Welcome to Interactive Data Visualization in R with Shiny. This is a step-by-step guide walking through the code we will write together during our hands-on session.

Learning Objectives

By the end of our two-part Shiny workshop, you should be able to:

  1. Articulate basic principles of interactive data visualization

  2. Understand critical concepts from R-Shiny including the structure of a Shiny App, how to run a Shiny App, how to configure the aesthetics of a Shiny App, and how the reactive programming model of Shiny works.

  3. Be able to augment a barebones Shiny app framework to develop a fully functioning Shiny app using provided data. This will encompass R and HTML programming.

  4. Be able to identify tools and resources for sharing and deploying a Shiny app. Be aware of advanced style configuration using CSS.

Step 7: Add Eruptions Histogram

library(shiny)
library(bslib)

# Define UI for app that draws a histogram ----
ui <- page_sidebar(# App title ----
                   title = "Hello Shiny", sidebar = sidebar(
                     # Input: Slider for the number of bins ----
                     
                     sliderInput(
                       inputId = "bins",
                       label = "Number of bins:",
                       min = 1,
                       max = 50,
                       value = 30
                     )),
                    plotOutput("distPlot"),
      # Output: Eruptions Histogram
      plotOutput(outputId = "distPlotEruptions")
                   )

# Define server logic required to draw a histogram ----
server <- function(input, output) {
  
  output$distPlot <- renderPlot({
    x <- faithful$waiting
    bins <- seq(min(x), max(x), length.out = input$bins + 1)
    
    hist(
      x,
      breaks = bins,
      col = "#75AADB",
      border = "white",
      xlab = "Waiting time to next eruption (in mins)",
      main = "Histogram of waiting times"
    )
  })
  
  
  output$distPlotEruptions <- renderPlot({
    
    x    <- faithful$eruptions
    bins <- seq(min(x), max(x), length.out = input$bins + 1)
    
    hist(x, breaks = bins, col = "#75AADB", border = "white",
         xlab = "Length of eruption (in mins)",
         main = "Histogram of eruption times")
    
  })
}

# Create Shiny app ----
shinyApp(ui = ui, server = server)

ui <- page_sidebar(

  sidebar = sidebar(
 
    # List selector based on the columns in a dataframe 
    varSelectInput(
      inputId = 'xvar',

      data = dplyr::select(penguins, where(is.numeric), -Year),

server <- function(input, output) {
  
  output$scatterplot <- renderPlot({
    
    
    ggplot(penguins,
           # Coerce a string to a literal using the !! prefix

               y = !!input$yvar,
               color = Species)) +
      geom_point(size = 5,
                 alpha = 0.5) +
      theme_minimal(base_size = 15) +

  })
  
}

Step 6: Species Selector

library(shiny)
library(bslib)
library(palmerpenguins)
library(ggplot2)

# load data
data(penguins, package = "palmerpenguins")

# remove missing values, in any row
penguins <- na.omit(penguins)

# rename the columns for clarity in app
colnames(penguins) <- c("Species",
                        "Island",
                        "Bill Length (mm)",
                        "Bill Depth (mm)",
                        "Flipper Length (mm)",
                        "Body Mass (g)",
                        "Sex",
                        "Year") 



ui <- page_sidebar(
  # App title
  title = 'Palmer Penguin Dashboard',
  sidebar = sidebar(
    
    varSelectInput(
      inputId = 'xvar', 
      label = 'X variable',
      data = dplyr::select(penguins, where(is.numeric), -Year),
      selected = 'Bill Length (mm)'
    ),
    
    varSelectInput(
      inputId = "yvar", 
      label = "Y variable",
      data = dplyr::select(penguins, where(is.numeric), -Year),
      selected = 'Bill Depth (mm)'
    ),

    # Checkbox selector - returns a vector of strings, length 0, 1, 
    # 2, or 3, depending on the number of species selected
    checkboxGroupInput(
      inputId = "species",
      label = "Select Penguin Species",
      choices = c("Adelie" = "Adelie",
                  "Chinstrap" = "Chinstrap",
                  "Gentoo" = "Gentoo"),
      selected = c("Adelie", "Chinstrap", "Gentoo")
    ),
    
    hr(), 
    "This application uses data from the package:",
    code('palmerpenguins'),
    img(src = 'penguins.png'),
    p("Artwork by @allison_horst")),
  
  
  card(card_header("Scatterplot"),
       plotOutput("scatterplot")),
 
  )


server <- function(input, output) {
  
 # Reactive function - returns a filtered dataframe
 penguins_filter <- reactive({
   req(input$species)
   penguins |> dplyr::filter(Species %in% input$species)
 })
 
  output$scatterplot <- renderPlot({
    
    
    ggplot(penguins_filter(),
           aes(x = !!input$xvar,
               y = !!input$yvar,
               color = Species)) +
      geom_point(size = 5,
                 alpha = 0.5) +
      theme_minimal(base_size = 15) +
      theme(legend.position = "bottom")
  })
  
}

shinyApp(ui, server)

server <- function(input, output, session) {
  
 penguins_filter <- reactive({
   req(input$species)
   penguins |> dplyr::filter(Species %in% input$species)
 })
 # Use aes_string to pass the variables programmatically 
 # - they are returned from varSelectInput as strings
  output$scatterplot <- renderPlot({
    
 # character and literal values in aes   
    penguins_filter <- penguins |> dplyr::filter(Species %in% input$species)
    
    ggplot(penguins_filter(),
           aes(x = !!input$xvar,
               y = !!input$yvar,
               color = Species)) +
      geom_point(size = 5,
                 alpha = 0.5) +
      {if (input$smoother) geom_smooth()} +
      {if (input$facet) facet_wrap(vars(Island))} +
      xlim(c(min(penguins[input$xvar]), 
             max(penguins[input$xvar]))) +
      ylim(c(min(penguins[input$yvar]), 
             max(penguins[input$yvar]))) +
      scale_color_manual(values = color_blind_palette) +
      theme_minimal(base_size = 15) +
      theme(legend.position = "bottom")
  })

  # Create output statistical summary of filtered dataframe  
  output$summary <- renderPrint({
    summary(penguins_filter())
  })

  # Create output table to display filtered dataframe in app 
  output$table <- renderTable(penguins_filter()) 
  
}

shinyApp(ui, server)
